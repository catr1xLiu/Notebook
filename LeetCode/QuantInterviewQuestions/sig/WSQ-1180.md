> [!example] Question 1180 · SIG
> ## Software Engineer Interview

Implement a stack data structure in a programming language of your choice (e.g., C\#), including all standard methods such as push(), pop(), and top(). The stack should accept only 12-bit integer numbers as input, and all exceptions should be handled properly. Complete this task within one hour.

[Source](https://www.thewallstreetquants.com/ebook#p=1181)

> [!success]- Solution (Click to expand)
> ```csharp
> using System;
>
> namespace CustomStack12Bit
> {
>     /// <summary>
>     /// A stack of signed 12-bit integers.
>     ///
>     /// Valid range:
>     ///     -2048 to 2047
>     ///
>     /// Supported operations:
>     ///     Push, Pop, Peek, Clear, Contains, Count, IsEmpty
>     /// </summary>
>     public class Stack12Bit
>     {
>         private const int MinValue = -2048;
>         private const int MaxValue = 2047;
>
>         private int[] _items;
>         private int _count;
>
>         public Stack12Bit(int capacity = 4)
>         {
>             if (capacity < 0)
>                 throw new ArgumentOutOfRangeException(nameof(capacity), "Capacity cannot be negative.");
>
>             _items = new int[capacity == 0 ? 4 : capacity];
>             _count = 0;
>         }
>
>         public int Count
>         {
>             get { return _count; }
>         }
>
>         public bool IsEmpty
>         {
>             get { return _count == 0; }
>         }
>
>         public void Push(int value)
>         {
>             Validate12Bit(value);
>
>             if (_count == _items.Length)
>             {
>                 Resize();
>             }
>
>             _items[_count] = value;
>             _count++;
>         }
>
>         public int Pop()
>         {
>             if (IsEmpty)
>                 throw new InvalidOperationException("Stack is empty. Cannot pop.");
>
>             _count--;
>             int value = _items[_count];
>             _items[_count] = 0; // Optional: clear reference/data
>             return value;
>         }
>
>         public int Top()
>         {
>             if (IsEmpty)
>                 throw new InvalidOperationException("Stack is empty. Cannot read top element.");
>
>             return _items[_count - 1];
>         }
>
>         public int Peek()
>         {
>             return Top();
>         }
>
>         public void Clear()
>         {
>             Array.Clear(_items, 0, _count);
>             _count = 0;
>         }
>
>         public bool Contains(int value)
>         {
>             for (int i = _count - 1; i >= 0; i--)
>             {
>                 if (_items[i] == value)
>                     return true;
>             }
>
>             return false;
>         }
>
>         private void Resize()
>         {
>             int newCapacity = _items.Length * 2;
>             if (newCapacity == 0)
>                 newCapacity = 4;
>
>             int[] newItems = new int[newCapacity];
>             Array.Copy(_items, newItems, _count);
>             _items = newItems;
>         }
>
>         private static void Validate12Bit(int value)
>         {
>             if (value < MinValue || value > MaxValue)
>             {
>                 throw new ArgumentOutOfRangeException(
>                     nameof(value),
>                     $"Value must be a signed 12-bit integer in range [{MinValue}, {MaxValue}]."
>                 );
>             }
>         }
>
>         public override string ToString()
>         {
>             if (IsEmpty)
>                 return "Stack is empty.";
>
>             string result = "Top -> ";
>             for (int i = _count - 1; i >= 0; i--)
>             {
>                 result += _items[i];
>                 if (i > 0)
>                     result += " ";
>             }
>
>             return result;
>         }
>     }
>
>     internal class Program
>     {
>         private static void Main()
>         {
>             try
>             {
>                 Stack12Bit stack = new Stack12Bit();
>
>                 stack.Push(100);
>                 stack.Push(-2048);
>                 stack.Push(2047);
>
>                 Console.WriteLine("Current stack:");
>                 Console.WriteLine(stack);
>
>                 Console.WriteLine("Top element: " + stack.Top());
>                 Console.WriteLine("Popped element: " + stack.Pop());
>                 Console.WriteLine("Top after pop: " + stack.Peek());
>                 Console.WriteLine("Count: " + stack.Count);
>                 Console.WriteLine("Contains 100: " + stack.Contains(100));
>                 Console.WriteLine("Contains 500: " + stack.Contains(500));
>
>                 // Uncomment to test exceptions:
>                 // stack.Push(3000);      // Out of 12-bit range
>                 // stack.Clear();
>                 // stack.Pop();           // Empty stack exception
>             }
>             catch (ArgumentOutOfRangeException ex)
>             {
>                 Console.WriteLine("Range error: " + ex.Message);
>             }
>             catch (InvalidOperationException ex)
>             {
>                 Console.WriteLine("Operation error: " + ex.Message);
>             }
>             catch (Exception ex)
>             {
>                 Console.WriteLine("Unexpected error: " + ex.Message);
>             }
>         }
>     }
> }
> ```
>
> ## Explanation
>
> A stack follows the $LIFO$ principle: last in, first out.
>
> ### Design
> The class `Stack12Bit` implements:
> - `Push(int value)`
> - `Pop()`
> - `Top()`
> - `Peek()`
> - `Clear()`
> - `Contains(int value)`
> - `Count`
> - `IsEmpty`
>
> ### 12-bit integer validation
> A signed $12$-bit integer has range:
>
> $$
> -2^{11} \text{ to } 2^{11} - 1
> $$
>
> that is,
>
> $$
> -2048 \text{ to } 2047
> $$
>
> So every pushed value is checked before insertion.
>
> ### Exception handling
> The implementation handles:
> - `ArgumentOutOfRangeException` when a value is outside the $12$-bit range
> - `InvalidOperationException` when attempting `Pop()` or `Top()` on an empty stack
> - A general `Exception` catch in `Main()` for unexpected failures
>
> ## Final Answer
>
> The above C# implementation provides a complete stack with standard methods and proper exception handling, while enforcing that all inserted values are signed $12$-bit integers in the range $[-2048, 2047]$.
