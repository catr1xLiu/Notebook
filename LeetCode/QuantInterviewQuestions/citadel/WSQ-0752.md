> [!example] Question 752 · Citadel
> ## Software Engineer Intern Interview

Implement a class for the game Connect 4.

[Source](https://www.thewallstreetquants.com/ebook#p=753)

> [!success]- Solution (Click to expand)
> ```python
> class Connect4:
>     ROWS = 6
>     COLS = 7
>     EMPTY = "."
>
>     def __init__(self):
>         self.board = [[self.EMPTY for _ in range(self.COLS)] for _ in range(self.ROWS)]
>         self.current_player = "X"
>         self.winner = None
>         self.moves_played = 0
>
>     def reset(self):
>         self.board = [[self.EMPTY for _ in range(self.COLS)] for _ in range(self.ROWS)]
>         self.current_player = "X"
>         self.winner = None
>         self.moves_played = 0
>
>     def display(self):
>         for row in self.board:
>             print(" ".join(row))
>         print("0 1 2 3 4 5 6")
>
>     def is_valid_move(self, col):
>         if not 0 <= col < self.COLS:
>             return False
>         return self.board[0][col] == self.EMPTY
>
>     def get_valid_moves(self):
>         return [col for col in range(self.COLS) if self.is_valid_move(col)]
>
>     def make_move(self, col):
>         if self.winner is not None:
>             raise ValueError("Game is already over.")
>         if not self.is_valid_move(col):
>             raise ValueError("Invalid move.")
>
>         for row in range(self.ROWS - 1, -1, -1):
>             if self.board[row][col] == self.EMPTY:
>                 self.board[row][col] = self.current_player
>                 self.moves_played += 1
>
>                 if self.check_winner(row, col):
>                     self.winner = self.current_player
>                 elif self.moves_played == self.ROWS * self.COLS:
>                     self.winner = "Draw"
>                 else:
>                     self.current_player = "O" if self.current_player == "X" else "X"
>                 return row, col
>
>     def check_winner(self, row, col):
>         player = self.board[row][col]
>         directions = [
>             (0, 1),   # horizontal
>             (1, 0),   # vertical
>             (1, 1),   # diagonal down-right
>             (1, -1),  # diagonal down-left
>         ]
>
>         for dr, dc in directions:
>             count = 1
>             count += self._count_direction(row, col, dr, dc, player)
>             count += self._count_direction(row, col, -dr, -dc, player)
>             if count >= 4:
>                 return True
>         return False
>
>     def _count_direction(self, row, col, dr, dc, player):
>         count = 0
>         r, c = row + dr, col + dc
>         while 0 <= r < self.ROWS and 0 <= c < self.COLS and self.board[r][c] == player:
>             count += 1
>             r += dr
>             c += dc
>         return count
>
>     def is_game_over(self):
>         return self.winner is not None
>
>     def get_winner(self):
>         return self.winner
>
>     def get_board(self):
>         return [row[:] for row in self.board]
> ```
