> ## Intern Interview

Write a function to traverse a binary tree without using recursion.

[Source](https://www.thewallstreetquants.com/ebook#p=746)

> [!success]- Solution (Click to expand)
> ## Binary Tree Traversal Without Recursion
>
> Tree traversal can be performed iteratively by using an explicit stack or queue instead of recursion. The three standard depth-first traversals are:
>
> - $Preorder$: root, left, right
> - $Inorder$: left, root, right
> - $Postorder$: left, right, root
>
> A level-order traversal uses a queue.
>
> Assume the binary tree node is defined as:
>
> ```cpp
> struct Node {
>     int data;
>     Node* left;
>     Node* right;
> };
> ```
>
> ---
>
> ## 1. Preorder Traversal Without Recursion
>
> ### Idea
> Visit the current node first, then process the left subtree, then the right subtree.  
> A stack is used so that the right child is pushed before the left child, ensuring the left subtree is processed first.
>
> ### Code
>
> ```cpp
> #include <iostream>
> #include <stack>
> using namespace std;
>
> void preorder(Node* root) {
>     if (root == nullptr) return;
>
>     stack<Node*> st;
>     st.push(root);
>
>     while (!st.empty()) {
>         Node* curr = st.top();
>         st.pop();
>
>         cout << curr->data << " ";
>
>         if (curr->right) st.push(curr->right);
>         if (curr->left) st.push(curr->left);
>     }
> }
> ```
>
> ---
>
> ## 2. Inorder Traversal Without Recursion
>
> ### Idea
> Move as far left as possible, pushing nodes onto the stack.  
> When no left child remains, pop a node, visit it, and then move to its right child.
>
> ### Code
>
> ```cpp
> #include <iostream>
> #include <stack>
> using namespace std;
>
> void inorder(Node* root) {
>     stack<Node*> st;
>     Node* curr = root;
>
>     while (curr != nullptr || !st.empty()) {
>         while (curr != nullptr) {
>             st.push(curr);
>             curr = curr->left;
>         }
>
>         curr = st.top();
>         st.pop();
>
>         cout << curr->data << " ";
>         curr = curr->right;
>     }
> }
> ```
>
> ---
>
> ## 3. Postorder Traversal Without Recursion
>
> ### Idea
> Postorder requires visiting a node after both subtrees.  
> One common iterative method uses two stacks:
>
> 1. Push the root into the first stack.
> 2. Pop from the first stack and push into the second stack.
> 3. Push the left and right children into the first stack.
> 4. Finally, pop all nodes from the second stack.
>
> This produces the order left, right, root.
>
> ### Code
>
> ```cpp
> #include <iostream>
> #include <stack>
> using namespace std;
>
> void postorder(Node* root) {
>     if (root == nullptr) return;
>
>     stack<Node*> st1, st2;
>     st1.push(root);
>
>     while (!st1.empty()) {
>         Node* curr = st1.top();
>         st1.pop();
>         st2.push(curr);
>
>         if (curr->left) st1.push(curr->left);
>         if (curr->right) st1.push(curr->right);
>     }
>
>     while (!st2.empty()) {
>         cout << st2.top()->data << " ";
>         st2.pop();
>     }
> }
> ```
>
> ---
>
> ## 4. Level-Order Traversal Without Recursion
>
> ### Idea
> Use a queue to process nodes level by level.
>
> ### Code
>
> ```cpp
> #include <iostream>
> #include <queue>
> using namespace std;
>
> void levelOrder(Node* root) {
>     if (root == nullptr) return;
>
>     queue<Node*> q;
>     q.push(root);
>
>     while (!q.empty()) {
>         Node* curr = q.front();
>         q.pop();
>
>         cout << curr->data << " ";
>
>         if (curr->left) q.push(curr->left);
>         if (curr->right) q.push(curr->right);
>     }
> }
> ```
>
> ---
>
> ## Complete Example
>
> ```cpp
> #include <iostream>
> #include <stack>
> #include <queue>
> using namespace std;
>
> struct Node {
>     int data;
>     Node* left;
>     Node* right;
>
>     Node(int val) : data(val), left(nullptr), right(nullptr) {}
> };
>
> void preorder(Node* root) {
>     if (root == nullptr) return;
>
>     stack<Node*> st;
>     st.push(root);
>
>     while (!st.empty()) {
>         Node* curr = st.top();
>         st.pop();
>
>         cout << curr->data << " ";
>
>         if (curr->right) st.push(curr->right);
>         if (curr->left) st.push(curr->left);
>     }
> }
>
> void inorder(Node* root) {
>     stack<Node*> st;
>     Node* curr = root;
>
>     while (curr != nullptr || !st.empty()) {
>         while (curr != nullptr) {
>             st.push(curr);
>             curr = curr->left;
>         }
>
>         curr = st.top();
>         st.pop();
>
>         cout << curr->data << " ";
>         curr = curr->right;
>     }
> }
>
> void postorder(Node* root) {
>     if (root == nullptr) return;
>
>     stack<Node*> st1, st2;
>     st1.push(root);
>
>     while (!st1.empty()) {
>         Node* curr = st1.top();
>         st1.pop();
>         st2.push(curr);
>
>         if (curr->left) st1.push(curr->left);
>         if (curr->right) st1.push(curr->right);
>     }
>
>     while (!st2.empty()) {
>         cout << st2.top()->data << " ";
>         st2.pop();
>     }
> }
>
> void levelOrder(Node* root) {
>     if (root == nullptr) return;
>
>     queue<Node*> q;
>     q.push(root);
>
>     while (!q.empty()) {
>         Node* curr = q.front();
>         q.pop();
>
>         cout << curr->data << " ";
>
>         if (curr->left) q.push(curr->left);
>         if (curr->right) q.push(curr->right);
>     }
> }
>
> int main() {
>     Node* root = new Node(1);
>     root->left = new Node(2);
>     root->right = new Node(3);
>     root->left->left = new Node(4);
>     root->left->right = new Node(5);
>
>     cout << "Preorder: ";
>     preorder(root);
>     cout << endl;
>
>     cout << "Inorder: ";
>     inorder(root);
>     cout << endl;
>
>     cout << "Postorder: ";
>     postorder(root);
>     cout << endl;
>
>     cout << "Level-order: ";
>     levelOrder(root);
>     cout << endl;
>
>     return 0;
> }
> ```
>

>
> ## Output for the Above Tree
>
> For the tree
>
> $$
> \begin{array}{c}
> 1 \\
> / \ \backslash \\
> 2 \quad 3 \\
> / \backslash \\
> 4 \quad 5
> \end{array}
> $$
>
> the traversals are:
>
> - $Preorder$: $1\ 2\ 4\ 5\ 3$
> - $Inorder$: $4\ 2\ 5\ 1\ 3$
> - $Postorder$: $4\ 5\ 2\ 3\ 1$
> - $Level\text{-}order$: $1\ 2\ 3\ 4\ 5$
>
> Thus, binary tree traversal without recursion is achieved by replacing the implicit function-call stack with an explicit stack or queue, depending on the traversal order.
