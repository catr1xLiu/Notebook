> [!example] Question 688 · Citadel
> ## Software Engineer Interview

Write code to implement a multi-reader/single-writer lock.

[Source](https://www.thewallstreetquants.com/ebook#p=689)

> [!success]- Solution (Click to expand)
> ```cpp
> #include <mutex>
> #include <condition_variable>
>
> class MultiReaderSingleWriterLock {
> public:
>     MultiReaderSingleWriterLock()
>         : active_readers_(0), waiting_writers_(0), writer_active_(false) {}
>
>     void lock_shared() {
>         std::unique_lock<std::mutex> lock(mutex_);
>         readers_cv_.wait(lock, [this] {
>             return !writer_active_ && waiting_writers_ == 0;
>         });
>         ++active_readers_;
>     }
>
>     void unlock_shared() {
>         std::unique_lock<std::mutex> lock(mutex_);
>         --active_readers_;
>         if (active_readers_ == 0) {
>             writers_cv_.notify_one();
>         }
>     }
>
>     void lock() {
>         std::unique_lock<std::mutex> lock(mutex_);
>         ++waiting_writers_;
>         writers_cv_.wait(lock, [this] {
>             return !writer_active_ && active_readers_ == 0;
>         });
>         --waiting_writers_;
>         writer_active_ = true;
>     }
>
>     void unlock() {
>         std::unique_lock<std::mutex> lock(mutex_);
>         writer_active_ = false;
>
>         if (waiting_writers_ > 0) {
>             writers_cv_.notify_one();
>         } else {
>             readers_cv_.notify_all();
>         }
>     }
>
> private:
>     std::mutex mutex_;
>     std::condition_variable readers_cv_;
>     std::condition_variable writers_cv_;
>     int active_readers_;
>     int waiting_writers_;
>     bool writer_active_;
> };
> ```
>
> ### Example usage
>
> ```cpp
> #include <iostream>
> #include <thread>
> #include <vector>
>
> MultiReaderSingleWriterLock rwlock;
> int shared_data = 0;
>
> void reader(int id) {
>     rwlock.lock_shared();
>     std::cout << "Reader " << id << " reads " << shared_data << '\n';
>     rwlock.unlock_shared();
> }
>
> void writer(int id) {
>     rwlock.lock();
>     ++shared_data;
>     std::cout << "Writer " << id << " writes " << shared_data << '\n';
>     rwlock.unlock();
> }
>
> int main() {
>     std::vector<std::thread> threads;
>
>     for (int i = 0; i < 3; ++i) {
>         threads.emplace_back(reader, i);
>     }
>
>     for (int i = 0; i < 2; ++i) {
>         threads.emplace_back(writer, i);
>     }
>
>     for (int i = 3; i < 6; ++i) {
>         threads.emplace_back(reader, i);
>     }
>
>     for (auto& t : threads) {
>         t.join();
>     }
>
>     return 0;
> }
> ```
>
> ### Properties
>
> - Multiple readers may hold the lock simultaneously.
> - Only one writer may hold the lock at a time.
> - Writers are given priority once waiting, preventing writer starvation.
