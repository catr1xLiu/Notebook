> [!example] Question 970 · Optiver
> ## Senior Software Engineer Interview

Create functions that simulate communication with exchanges and ensure that short-selling is avoided.

[Source](https://www.thewallstreetquants.com/ebook#p=971)

> [!success]- Solution (Click to expand)
> ```python
> from dataclasses import dataclass, field
> from typing import Dict, List, Optional
> import time
> import uuid
>
>
> @dataclass
> class Order:
>     id: str
>     symbol: str
>     side: str          # "BUY" or "SELL"
>     quantity: int
>     price: Optional[float] = None
>     status: str = "NEW"
>     timestamp: float = field(default_factory=time.time)
>
>
> class ExchangeSimulator:
>     """
>     Simulates a simple exchange:
>     - accepts market and limit orders
>     - fills market orders immediately at current market price
>     - fills limit orders if marketable, otherwise leaves them open
>     """
>
>     def __init__(self, name: str, market_prices: Optional[Dict[str, float]] = None):
>         self.name = name
>         self.market_prices = market_prices or {}
>         self.orders: Dict[str, Order] = {}
>
>     def set_price(self, symbol: str, price: float) -> None:
>         self.market_prices[symbol] = price
>
>     def get_price(self, symbol: str) -> float:
>         if symbol not in self.market_prices:
>             raise ValueError(f"No market price set for {symbol}")
>         return self.market_prices[symbol]
>
>     def place_market_order(self, symbol: str, side: str, quantity: int) -> Order:
>         if quantity <= 0:
>             raise ValueError("quantity must be positive")
>
>         price = self.get_price(symbol)
>         order = Order(
>             id=str(uuid.uuid4()),
>             symbol=symbol,
>             side=side.upper(),
>             quantity=quantity,
>             price=price,
>             status="FILLED",
>         )
>         self.orders[order.id] = order
>         return order
>
>     def place_limit_order(self, symbol: str, side: str, quantity: int, limit_price: float) -> Order:
>         if quantity <= 0:
>             raise ValueError("quantity must be positive")
>
>         side = side.upper()
>         market_price = self.get_price(symbol)
>
>         marketable = (
>             (side == "BUY" and limit_price >= market_price) or
>             (side == "SELL" and limit_price <= market_price)
>         )
>
>         order = Order(
>             id=str(uuid.uuid4()),
>             symbol=symbol,
>             side=side,
>             quantity=quantity,
>             price=limit_price,
>             status="FILLED" if marketable else "OPEN",
>         )
>         self.orders[order.id] = order
>         return order
>
>     def cancel_order(self, order_id: str) -> Order:
>         if order_id not in self.orders:
>             raise ValueError("order not found")
>
>         order = self.orders[order_id]
>         if order.status == "OPEN":
>             order.status = "CANCELED"
>         return order
>
>     def get_order(self, order_id: str) -> Order:
>         if order_id not in self.orders:
>             raise ValueError("order not found")
>         return self.orders[order_id]
>
>
> class NoShortSellingBroker:
>     """
>     Broker layer enforcing:
>     - no short selling
>     - positions may not go below zero
>     - sell quantity must be <= current position
>     """
>
>     def __init__(self, exchange: ExchangeSimulator, starting_cash: float = 0.0):
>         self.exchange = exchange
>         self.cash = starting_cash
>         self.positions: Dict[str, int] = {}
>         self.trade_log: List[Order] = []
>
>     def get_position(self, symbol: str) -> int:
>         return self.positions.get(symbol, 0)
>
>     def get_cash(self) -> float:
>         return self.cash
>
>     def _apply_filled_order(self, order: Order) -> None:
>         if order.status != "FILLED":
>             return
>
>         symbol = order.symbol
>         qty = order.quantity
>         px = order.price if order.price is not None else self.exchange.get_price(symbol)
>
>         if order.side == "BUY":
>             self.positions[symbol] = self.get_position(symbol) + qty
>             self.cash -= qty * px
>         elif order.side == "SELL":
>             self.positions[symbol] = self.get_position(symbol) - qty
>             self.cash += qty * px
>
>         self.trade_log.append(order)
>
>     def buy_market(self, symbol: str, quantity: int) -> Order:
>         order = self.exchange.place_market_order(symbol, "BUY", quantity)
>         self._apply_filled_order(order)
>         return order
>
>     def sell_market(self, symbol: str, quantity: int) -> Order:
>         current_position = self.get_position(symbol)
>         if quantity > current_position:
>             raise ValueError(
>                 f"Short selling not allowed: tried to sell {quantity}, "
>                 f"but only hold {current_position} shares of {symbol}"
>             )
>
>         order = self.exchange.place_market_order(symbol, "SELL", quantity)
>         self._apply_filled_order(order)
>         return order
>
>     def buy_limit(self, symbol: str, quantity: int, limit_price: float) -> Order:
>         order = self.exchange.place_limit_order(symbol, "BUY", quantity, limit_price)
>         self._apply_filled_order(order)
>         return order
>
>     def sell_limit(self, symbol: str, quantity: int, limit_price: float) -> Order:
>         current_position = self.get_position(symbol)
>         if quantity > current_position:
>             raise ValueError(
>                 f"Short selling not allowed: tried to sell {quantity}, "
>                 f"but only hold {current_position} shares of {symbol}"
>             )
>
>         order = self.exchange.place_limit_order(symbol, "SELL", quantity, limit_price)
>         self._apply_filled_order(order)
>         return order
>
>     def cancel_order(self, order_id: str) -> Order:
>         return self.exchange.cancel_order(order_id)
>
>     def portfolio(self) -> Dict[str, int]:
>         return dict(self.positions)
>
>     def account_summary(self) -> Dict[str, object]:
>         return {
>             "cash": self.cash,
>             "positions": dict(self.positions),
>             "trades": len(self.trade_log),
>         }
>
>
> # Example usage
> if __name__ == "__main__":
>     exchange = ExchangeSimulator("DemoExchange", {"AAPL": 180.0, "MSFT": 410.0})
>     broker = NoShortSellingBroker(exchange, starting_cash=10000.0)
>
>     broker.buy_market("AAPL", 10)
>     broker.sell_market("AAPL", 4)
>
>     try:
>         broker.sell_market("AAPL", 20)  # rejected: would short-sell
>     except ValueError as e:
>         print("Rejected:", e)
>
>     print(broker.account_summary())
> ```
